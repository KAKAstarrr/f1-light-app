## 二、表结构

### 2.1 users — 用户表

```sql
CREATE TABLE users (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,  -- PG: SERIAL PRIMARY KEY
    username        VARCHAR(50)  NOT NULL UNIQUE,
    email           VARCHAR(120) NOT NULL UNIQUE,
    password_hash   VARCHAR(128) NOT NULL,              -- bcrypt hash
    display_name    VARCHAR(100),
    avatar_url      VARCHAR(500),
    role            VARCHAR(20)  DEFAULT 'user',        -- user / admin
    created_at      TIMESTAMP    DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP    DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_username ON users(username);
```

### 2.2 drivers — 车手表

```sql
CREATE TABLE drivers (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    driver_ref      VARCHAR(50)  NOT NULL UNIQUE,       -- Ergast driverId, 如 "max_verstappen"
    name            VARCHAR(100) NOT NULL,               -- 全名 "Max Verstappen"
    code            CHAR(3),                             -- 三字母 "VER"
    number          INTEGER,                             -- 车号
    nationality     VARCHAR(50),
    current_constructor VARCHAR(100),
    date_of_birth   DATE,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE UNIQUE INDEX idx_drivers_ref ON drivers(driver_ref);
CREATE INDEX idx_drivers_code ON drivers(code);
```

### 2.3 constructors — 车队表

```sql
CREATE TABLE constructors (
    id                  INTEGER PRIMARY KEY AUTOINCREMENT,
    constructor_ref     VARCHAR(50)  NOT NULL UNIQUE,    -- "red_bull"
    name                VARCHAR(100) NOT NULL,            -- "Red Bull Racing"
    nationality         VARCHAR(50),
    created_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE UNIQUE INDEX idx_constructors_ref ON constructors(constructor_ref);
```

### 2.4 fantasy_teams — Fantasy 阵容表

```sql
CREATE TABLE fantasy_teams (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id         INTEGER NOT NULL REFERENCES users(id),
    round           INTEGER NOT NULL,
    chip_used       VARCHAR(20) DEFAULT 'none',          -- none / limitless / wildcard / no_negative
    total_cost      DECIMAL(8,2) NOT NULL,               -- 阵容总花费（M）
    is_submitted    BOOLEAN DEFAULT FALSE,               -- 是否已提交
    submitted_at    TIMESTAMP,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_fantasy_teams_user_round ON fantasy_teams(user_id, round);
CREATE UNIQUE INDEX idx_fantasy_teams_unique ON fantasy_teams(user_id, round);
-- 每人每站只能有一份阵容
```

### 2.5 fantasy_team_drivers — 阵容-车手关联

```sql
CREATE TABLE fantasy_team_drivers (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    team_id         INTEGER NOT NULL REFERENCES fantasy_teams(id) ON DELETE CASCADE,
    driver_id       INTEGER NOT NULL REFERENCES drivers(id),
    is_captain      BOOLEAN DEFAULT FALSE,               -- x2 Boost
    price           DECIMAL(8,2) NOT NULL,               -- 选取时的价格
    UNIQUE(team_id, driver_id)
);

CREATE INDEX idx_ftd_team ON fantasy_team_drivers(team_id);
```

### 2.6 fantasy_team_constructors — 阵容-车队关联

```sql
CREATE TABLE fantasy_team_constructors (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    team_id         INTEGER NOT NULL REFERENCES fantasy_teams(id) ON DELETE CASCADE,
    constructor_id  INTEGER NOT NULL REFERENCES constructors(id),
    price           DECIMAL(8,2) NOT NULL,
    UNIQUE(team_id, constructor_id)
);

CREATE INDEX idx_ftc_team ON fantasy_team_constructors(team_id);
```

### 2.7 fantasy_scores — Fantasy 积分明细

```sql
CREATE TABLE fantasy_scores (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id         INTEGER NOT NULL REFERENCES users(id),
    round           INTEGER NOT NULL,
    team_id         INTEGER REFERENCES fantasy_teams(id),
    total_points    INTEGER NOT NULL,
    detail          TEXT NOT NULL,                        -- JSON: 积分明细
    chip_effect_applied BOOLEAN DEFAULT FALSE,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- detail JSON 结构示例:
-- {
--   "drivers": [{"name":"VER","points":20,"captain_bonus":20,"breakdown":{...}}],
--   "constructors": [{"name":"Red Bull","points":5}],
--   "bonuses": {"fastest_lap":5,"position_gain":3},
--   "penalties": {"dnf":-5,"transfer_excess":-10}
-- }

CREATE INDEX idx_fs_user_round ON fantasy_scores(user_id, round);
CREATE UNIQUE INDEX idx_fs_unique ON fantasy_scores(user_id, round);
```

### 2.8 leagues — 联盟表

```sql
CREATE TABLE leagues (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    name            VARCHAR(100) NOT NULL,
    description     TEXT,
    creator_id      INTEGER NOT NULL REFERENCES users(id),
    invite_code     VARCHAR(20) UNIQUE,
    is_public       BOOLEAN DEFAULT FALSE,
    max_members     INTEGER DEFAULT 50,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_leagues_creator ON leagues(creator_id);
CREATE UNIQUE INDEX idx_leagues_invite ON leagues(invite_code);
```

### 2.9 league_members — 联盟成员

```sql
CREATE TABLE league_members (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    league_id       INTEGER NOT NULL REFERENCES leagues(id) ON DELETE CASCADE,
    user_id         INTEGER NOT NULL REFERENCES users(id),
    joined_at       TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(league_id, user_id)
);

CREATE INDEX idx_lm_league ON league_members(league_id);
CREATE INDEX idx_lm_user ON league_members(user_id);
```

### 2.10 predictions — AI 预测记录

```sql
CREATE TABLE predictions (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    round           INTEGER NOT NULL,
    driver_id       INTEGER NOT NULL REFERENCES drivers(id),
    probability     DECIMAL(8,6) NOT NULL,               -- 夺冠概率 0-1
    win_probability DECIMAL(8,6),                        -- 夺冠概率（胜利）
    rank_pred       INTEGER NOT NULL,                    -- AI 预测排名
    model_version   VARCHAR(20) NOT NULL,                -- 模型版本号
    features_json   TEXT,                                -- 该预测使用的特征快照
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_pred_round ON predictions(round);
CREATE INDEX idx_pred_driver ON predictions(driver_id);
CREATE INDEX idx_pred_round_driver ON predictions(round, driver_id);
```

### 2.11 votes — 投票表

```sql
CREATE TABLE votes (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id         INTEGER NOT NULL REFERENCES users(id),
    round           INTEGER NOT NULL,
    driver_id       INTEGER NOT NULL REFERENCES drivers(id),
    vote_type       VARCHAR(20) DEFAULT 'driver_of_day',  -- driver_of_day / season_champion / constructor_champion
    voted_at        TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_votes_round ON votes(round);
CREATE INDEX idx_votes_round_type ON votes(round, vote_type);
CREATE UNIQUE INDEX idx_votes_unique ON votes(user_id, round, vote_type);
-- 每人每站每种投票只能投一次
```

### 2.12 leaderboards — 排行榜快照

```sql
CREATE TABLE leaderboards (
    id                  INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id             INTEGER NOT NULL REFERENCES users(id),
    round               INTEGER,                         -- NULL 表示赛季总榜
    season              INTEGER NOT NULL,
    fantasy_score       INTEGER NOT NULL,
    rank                INTEGER NOT NULL,
    detail_json         TEXT,                            -- 得分明细 JSON
    snapshot_at         TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_lb_season_round ON leaderboards(season, round);
CREATE INDEX idx_lb_user_season ON leaderboards(user_id, season);
```

### 2.13 sessions — 比赛元数据快照（可选）

```sql
CREATE TABLE sessions (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    year            INTEGER NOT NULL,
    round           INTEGER NOT NULL,
    race_name       VARCHAR(200),
    circuit_name    VARCHAR(200),
    session_date    DATE,
    weather_json    TEXT,                                -- 天气快照 JSON

```

## 三、索引策略

| 表 | 索引 | 类型 | 用途 |

| users | `idx_users_email` | B-tree UNIQUE | 登录查询 |

### 4.3 轻量自动迁移（当前 SQLite 方案，2026-08-12 新增）

> 背景：SQLite `create_all` **只建新表、不补旧表新列**。ORM 模型给 `users` 表新增 3 个芯片字段后，旧库登录/注册直接报 `no such column: users.chip_limitless_used` → 500。

`backend/database.py` 的 `_auto_migrate()` 机制：

```python
def _auto_migrate():
    """启动时对比 ORM 模型与库表，自动 ALTER TABLE 补缺失列（保留数据）"""
    inspector = inspect(engine)
    for table_name, table in Base.metadata.tables.items():
        if not inspector.has_table(table_name):
            continue  # 新表交给 create_all
        db_cols = {c["name"] for c in inspector.get_columns(table_name)}
        for col in table.columns:
            if col.name not in db_cols:
                ddl = f'ALTER TABLE {table_name} ADD COLUMN {col.name} {col.type.compile(dialect=engine.dialect)}'
                engine.execute(text(ddl))
```

- 在 `init_db()` 中先执行 `_auto_migrate()` 再 `create_all()`
- 实测自动补 4 列（users 3 个芯片字段 + fantasy_teams.transfers_used），数据零丢失
- **约定**：SQLite 开发阶段用此机制；上 PostgreSQL 后切换 Alembic 正式版本管理

## 五、关键业务逻辑 SQL

### 5.1 Fantasy 积分结算查询

```sql
-- 获取某用户在某分站的 Fantasy 积分明细
SELECT 
    u.username,
    fs.total_points,
    fs.detail
FROM fantasy_scores fs
JOIN users u ON fs.user_id = u.id
WHERE fs.round = 3
ORDER BY fs.total_points DESC
LIMIT 20;
```

### 5.2 联盟排行榜

```sql
SELECT 
    lm.league_id,
    u.username,
    SUM(fs.total_points) AS season_total,
    RANK() OVER (PARTITION BY lm.league_id ORDER BY SUM(fs.total_points) DESC) AS rank
FROM league_members lm
JOIN fantasy_scores fs ON lm.user_id = fs.user_id
JOIN users u ON lm.user_id = u.id
WHERE lm.league_id = 1
GROUP BY lm.league_id, u.username
ORDER BY season_total DESC;
```

### 5.3 投票实时统计

```sql
SELECT 
    d.code,
    d.name,
    COUNT(*) AS vote_count,
    ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER(), 1) AS percentage
FROM votes v
JOIN drivers d ON v.driver_id = d.id
WHERE v.round = 2 AND v.vote_type = 'driver_of_day'
GROUP BY d.code, d.name
ORDER BY vote_count DESC;
```

### 5.4 AI 预测准确率回溯

```sql
-- 需要与实际赛果表 (race_results) Join
-- 此处假设有 race_results 表
SELECT 
    p.round,
    p.model_version,
    COUNT(*) AS total_predictions,
    SUM(CASE WHEN p.rank_pred <= 3 AND rr.position <= 3 THEN 1 ELSE 0 END) AS top3_correct
FROM predictions p
JOIN race_results rr ON p.round = rr.round AND p.driver_id = rr.driver_id
WHERE p.model_version = 'v2.1'
GROUP BY p.round, p.model_version;
```

## 六、简历话术提炼

### 6.1 数据库设计能力
> "设计 11 张核心数据表（用户/车手/车队/Fantasy阵容/积分/联盟/预测/投票/排行榜），涵盖用户体系、Fantasy 游戏机制、AI 预测记录、社区互动四大业务域，通过外键约束 + 联合唯一索引保障数据一致性。"

### 6.2 性能优化能力
> "针对 Fantasy 排行榜热点查询设计复合索引 (season, round)，对预测表建立 (round, driver_id) 联合索引，预估单表百万级数据下查询 < 50ms；使用 PostgreSQL 表分区按赛季/分站水平拆分大表。"

### 6.3 数据工程能力
> "Fantasy 积分结算采用 detail 字段存储 JSON 格式明细，既保留结构化查询能力（total_points 索引排序），又提供灵活的半结构化明细追溯，避免过度范式化导致大量 Join。"
